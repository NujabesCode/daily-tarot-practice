## "보이는 카드 6장" 만들기

화면 폭을 `W`라고 하면

```
카드 폭   = W / 5
카드 간격 = W / 6   (다음 카드까지의 거리)
→ 카드끼리 W/30 만큼 겹치고, 한 화면에 약 6장이 보임
→ 9장이니 나머지 3장은 옆으로 밀어서 봄
```

`LazyRow`의 `spacedBy`에 **음수 간격**(`W/6 - W/5`)을 주면 카드가 겹칩니다.

## 부채꼴 모양

카드마다 "화면 가운데에서 얼마나 떨어져 있나" (`f`, -1 ~ 1)를 구해서

```
rotationZ    = f × 14도          왼쪽 카드는 왼쪽으로, 오른쪽 카드는 오른쪽으로 기울기
translationY = |f| × 30dp        가장자리일수록 아래로
회전 기준점   = 카드 아래 가운데
```

스크롤하면 `f`가 바뀌어서 **부채가 따라 움직입니다.**

## 따라하기

새 파일 `LoveTarotScreen.kt`

```kotlin
package com.example.dailytarot

import androidx.compose.foundation.Image
import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.Arrangement
import androidx.compose.foundation.layout.BoxWithConstraints
import androidx.compose.foundation.layout.Column
import androidx.compose.foundation.layout.PaddingValues
import androidx.compose.foundation.layout.Spacer
import androidx.compose.foundation.layout.aspectRatio
import androidx.compose.foundation.layout.fillMaxSize
import androidx.compose.foundation.layout.fillMaxWidth
import androidx.compose.foundation.layout.height
import androidx.compose.foundation.layout.width
import androidx.compose.foundation.lazy.LazyRow
import androidx.compose.foundation.lazy.itemsIndexed
import androidx.compose.foundation.lazy.rememberLazyListState
import androidx.compose.foundation.shape.RoundedCornerShape
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.runtime.remember
import androidx.compose.ui.Alignment
import androidx.compose.ui.Modifier
import androidx.compose.ui.draw.clip
import androidx.compose.ui.draw.shadow
import androidx.compose.ui.graphics.Color
import androidx.compose.ui.graphics.TransformOrigin
import androidx.compose.ui.graphics.graphicsLayer
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.platform.LocalContext
import androidx.compose.ui.res.painterResource
import androidx.compose.ui.text.font.FontWeight
import androidx.compose.ui.text.style.TextAlign
import androidx.compose.ui.unit.dp
import androidx.compose.ui.unit.sp
import kotlin.math.abs

@Composable
fun LoveTarotScreen(onPick: (TarotCard) -> Unit) {
    val context = LocalContext.current
    val deck = remember { TarotData.load(context, TarotKind.Love.file).shuffled() }   // 9장 랜덤
    val listState = rememberLazyListState()

    TarotBackground(showMoon = false, topCloud = true, bottomCloud = false) {
        BoxWithConstraints(Modifier.fillMaxSize()) {
            val cardW = maxWidth / 5
            val step = maxWidth / 6                       // 화면에 6장

            // 위쪽 문구
            Column(Modifier.fillMaxWidth(), horizontalAlignment = Alignment.CenterHorizontally) {
                Spacer(Modifier.height(40.dp))
                LogoMark()
                Spacer(Modifier.height(8.dp))
                Text(
                    "인연 타로를 선택하셨네요.\n신중하게 카드 1장을 선택해주세요.",
                    color = Color.White,
                    fontFamily = Myeongjo,
                    fontWeight = FontWeight.Bold,
                    fontSize = 17.sp,
                    lineHeight = 24.sp,
                    textAlign = TextAlign.Center
                )
                Spacer(Modifier.height(10.dp))
                Text("생각하고 있는 그 사람과 인연이 될 수 있을까요?", color = White70, fontFamily = NotoSans, fontSize = 12.sp)
            }

            // 가운데 안내
            Text(
                "좌우로 스크롤하여\n카드 한 장을 골라보세요!",
                color = White70,
                fontFamily = NotoSans,
                fontSize = 16.sp,
                lineHeight = 30.sp,
                textAlign = TextAlign.Center,
                modifier = Modifier.align(Alignment.Center)
            )

            // 아래 부채꼴 카드
            LazyRow(
                state = listState,
                horizontalArrangement = Arrangement.spacedBy(step - cardW),   // 음수 = 겹치기
                verticalAlignment = Alignment.Bottom,
                contentPadding = PaddingValues(horizontal = 8.dp),
                modifier = Modifier
                    .align(Alignment.BottomCenter)
                    .fillMaxWidth()
                    .height(cardW * 1.6f + 60.dp)
            ) {
                itemsIndexed(deck) { index, card ->
                    Image(
                        painter = painterResource(R.drawable.tarot_card_back),
                        contentDescription = "카드 ${index + 1}",
                        contentScale = ContentScale.FillBounds,
                        modifier = Modifier
                            .width(cardW)
                            .aspectRatio(400f / 640f)
                            .graphicsLayer {
                                val layout = listState.layoutInfo
                                val info = layout.visibleItemsInfo.firstOrNull { it.index == index }
                                val center = layout.viewportEndOffset / 2f
                                val itemCenter = if (info != null) info.offset + info.size / 2f else center
                                val f = ((itemCenter - center) / center).coerceIn(-1.5f, 1.5f)
                                transformOrigin = TransformOrigin(0.5f, 1f)
                                rotationZ = f * 14f
                                translationY = abs(f) * 30.dp.toPx() + 24.dp.toPx()
                            }
                            .shadow(6.dp, RoundedCornerShape(6.dp))
                            .clip(RoundedCornerShape(6.dp))
                            .clickable { onPick(card) }
                    )
                }
            }
        }
    }
}
```

## 핵심 설명

- **`graphicsLayer` 안에서 `listState.layoutInfo`를 읽기**: 스크롤할 때마다 이 블록만 다시 계산되어 부드럽습니다. (화면 전체를 다시 그리지 않음)
- **`transformOrigin = TransformOrigin(0.5f, 1f)`**: 카드 **아래 가운데**를 축으로 돌려서 손에 쥔 카드처럼 펼쳐집니다.
- **겹친 카드 클릭**: 나중에 그려진(오른쪽) 카드가 위에 있으므로 겹친 부분을 누르면 오른쪽 카드가 선택됩니다.
- **구름은 위만**: 채점표 UI 요소가 "구름 그래픽 이미지 (위 배경)"이므로 `bottomCloud = false`.
